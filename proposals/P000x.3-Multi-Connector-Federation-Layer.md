🔀 四、Federation Router（调度器）
federation-router.ts
import { getConnector } from '#app/connectors/registry/connector-registry.js';
import type { FederationRequest } from './federation-types.js';

export const routeFederation = async (
  req: FederationRequest,
  shopId: string,
  date: string,
) => {
  const tasks = req.platforms.map((platform) => {
    const connector = getConnector(platform);

    return connector.acquire(shopId, [], {
      date,
      mock: false,
    });
  });

  const results = await Promise.allSettled(tasks);

  return results.map((r, i) => ({
    platform: req.platforms[i],
    success: r.status === 'fulfilled',
    data: r.status === 'fulfilled' ? r.value : null,
  }));
};
📊 五、Execution Aggregator（核心聚合层）
execution-aggregator.ts
import type { PlatformExecutionResult } from './federation-types.js';

export const aggregateResults = (results: PlatformExecutionResult[]) => {
  const success = results.filter(r => r.success);

  return {
    platform_count: results.length,
    success_count: success.length,

    merged: {
      summary: success.map(r => (r.data as any)?.summary),
      trends: success.map(r => (r.data as any)?.trend),
      products: success.map(r => (r.data as any)?.productTop),
    },

    conflicts: results.filter(r => !r.success),
  };
};
🧠 六、Federation Kernel（主入口）
federation-kernel.ts
import { routeFederation } from './federation-router.js';
import { aggregateResults } from './execution-aggregator.js';

export const executeFederation = async (req: {
  shop_id: string;
  date: string;
  platforms: string[];
}) => {
  const results = await routeFederation(
    {
      shop_id: req.shop_id,
      date: req.date,
      platforms: req.platforms,
      mode: 'merge',
    },
    req.shop_id,
    req.date,
  );

  const aggregated = aggregateResults(results);

  return {
    request: req,
    results,
    aggregated,
  };
};

🧠 本阶段本质

P0006.3 = “把系统从单数据源升级为数据联邦系统”

Introduce a federation layer that can execute multiple connectors in parallel and aggregate results into a unified response.
Do not hardcode platform logic in CLI or kernel. All platform execution must go through connector registry and federation router.
