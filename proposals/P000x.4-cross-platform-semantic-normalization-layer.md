🔁 五、Semantic Mapper（核心转换器）
semantic-mapper.ts
import { SEMANTIC_METRICS } from './semantic-registry.js';

export const buildSemanticMap = (platformMap: Record<string, string>) => {
  const reverseIndex = new Map<string, string>();

  for (const metric of SEMANTIC_METRICS) {
    for (const alias of metric.aliases) {
      reverseIndex.set(alias, metric.canonical_name);
    }
  }

  const result: Record<string, string> = {};

  for (const [raw, value] of Object.entries(platformMap)) {
    const canonical = reverseIndex.get(raw);
    if (canonical) {
      result[canonical] = value;
    }
  }

  return result;
};
🧠 六、Semantic Resolver（运行时统一入口）
