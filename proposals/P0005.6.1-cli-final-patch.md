🟢 collect CLI 新实现
const cmdCollect = async (args: string[]): Promise<void> => {
  const { values, positionals } = parseArgs({
    args,
    options: {
      mode: { type: 'string', default: 'mock' },
      date: { type: 'string' },
      days: { type: 'string', default: '1' },
    },
    allowPositionals: true,
  });

  const [source, shopId] = positionals as string[] | [];
  if (!source || !shopId) {
    console.error('Usage: cli collect <source> <shopId>');
    process.exit(1);
  }

  if (source !== 'jd') {
    console.error(`Only JD supported`);
    process.exit(1);
  }

  const kernel = getKernel();

  const result = await kernel.execute({
    platform: 'jd',
    shopId,
    mode: values.mode === 'live' ? 'live' : 'mock',
    date: values.date,
    days: Number(values.days ?? '1'),
  });

  console.log(`[kernel] execution completed`);
  console.log(`success: ${result.success}`);
  console.log(`signals: ${result.signals?.length ?? 0}`);
  console.log(`evidence: ${result.evidence?.length ?? 0}`);

  openDb().close();
};
