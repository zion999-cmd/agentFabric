2️⃣ connector-registry.ts
import type { Connector } from './connector-types.js';

const registry = new Map<string, Connector>();

export const registerConnector = (c: Connector) => {
  registry.set(c.platform, c);
};

export const getConnector = (platform: string): Connector => {
  const c = registry.get(platform);
  if (!c) throw new Error(`Connector not found: ${platform}`);
  return c;
};

export const listConnectors = () => [...registry.keys()];
3️⃣ connector-factory.ts
import { registerConnector } from './connector-registry.js';
import type { Connector } from './connector-types.js';

export const createConnector = (connector: Connector) => {

  return connector;
};

import { createConnector } from '../registry/connector-factory.js';
import { acquireJdData } from './acquisition/index.js';

export const jdConnector = createConnector({
  platform: 'jd',
  acquire: async (shopId, endpoints, options) => {
    const res = await acquireJdData({
      shopId,
      mock: options?.mock ?? true,
      fromDate: options?.date,
      toDate: options?.date,
    });

    if (!res.success) throw new Error('JD acquire failed');

    return {
      summary: res.data?.summary,
      trend: res.data?.hourly_gmv,
      productTop: res.data?.top_products,
    };
  },
});
