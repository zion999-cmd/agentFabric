    var margin = -15;

    function n(v) {
      return Math.round(v * 10000) / 10000;
    }
    function px(value) {
      return typeof value === 'number' ? (value === 0 ? 0 : (Math.round(n(value)) + 'px')) : value;
    }
    var getMainDirectionPos = function (
        mainTriggerPos,
        mainTriggerSize,
        mainPopoverSize,
        mainClientSize,
        directions,
    ) {
        var adjustSize = mainPopoverSize; // 自适应尺寸

        var pos = [
            mainTriggerPos - margin - mainPopoverSize,
            mainTriggerPos + mainTriggerSize + margin,
        ];
        var space = [
            mainTriggerPos - margin,
            mainClientSize - (mainTriggerPos + mainTriggerSize + margin),
        ];
        var realIndex = directions.findIndex(x => x === mainDirection);
        var opposedIndex = 1 - realIndex;

        // 如果本侧空间不够，而对侧空间能容纳弹出内容，使用对侧控件
        if (autoAdjustOverflow === 'auto' || autoAdjustOverflow === 'main') {
            if (space[realIndex] < adjustSize && space[opposedIndex] >= adjustSize) {
                realIndex = opposedIndex;
                realMainDirection = directions[realIndex];
                // console.log('realMainDirection', realMainDirection);
            }
        }
        return px(pos[realIndex]);
    };
    // 计算次方向的位置
    var getSubDirectionPos = function (
        subTriggerPos,
        subTriggerSize,
        subPopoverSize,
        subClientSize,
        subAlign
    ) {
        var getArrowPos = align => (subPopoverSize - 40 - arrowHalfSize * 2) * align + 20;
        var getPos = align => subTriggerPos + (subTriggerSize - subPopoverSize) * align;

        var pos;
        var arrowPos = getArrowPos(subAlign);

        pos = getPos(subAlign);

        function setNewPos(nextPos) {
            pos = nextPos;
        }

        // 移动到屏幕内
        if (autoAdjustOverflow === 'auto' || autoAdjustOverflow === 'sub') {
            if (pos < 0) {
                setNewPos(0);
            } else if (pos + subPopoverSize > subClientSize) {
                setNewPos(Math.floor(subClientSize - subPopoverSize));
            }
        }

        return {
            pos: px(pos),
            arrowPos: px(arrowPos),
        };
    };
    function popoverGenerator(triggerNode, portal, popoverRef, open) {
      if (open && triggerNode.getBoundingClientRect) {
        var subAlign = 0.5; // 次位置对齐
        if (subDirection) {
            subAlign = {
                Left: 0,
                Right: 1,
                Top: 0,
                Bottom: 1,
            }[subDirection];
        }

        var triggerNodeRect = triggerNode.getBoundingClientRect();
        var triggerRawX = triggerNodeRect.x,
          triggerRawY = triggerNodeRect.y,
          triggerWidth = triggerNodeRect.width,
          triggerHeight = triggerNodeRect.height;
        var protalRect = portal.getBoundingClientRect(triggerNode); // portal在特殊情况下不一定在页面右上，比如在shadow-dom中
        var rootX = protalRect.x, rootY = protalRect.y;
        var triggerX = triggerRawX - rootX, triggerY = triggerRawY - rootY; // 计算triggerNode的相对于portal的位置

        var popoverRefStyle = window.getComputedStyle(popoverRef);
        var styleWidth = popoverRefStyle.width, styleHeight = popoverRefStyle.height;
        var popoverWidth = parseFloat(styleWidth), popoverHeight = parseFloat(styleHeight);
        var clientWidth = document.documentElement.clientWidth, clientHeight = document.documentElement.clientHeight;
        var popoverStyle = {}, arrowStyle = {};

        var subDirectionPos = getSubDirectionPos(
            triggerX,
            triggerWidth,
            popoverWidth,
            clientWidth,
            subAlign
        );
        var pos = subDirectionPos.pos, arrowPos = subDirectionPos.arrowPos;
        popoverStyle.top = getMainDirectionPos(
            triggerY,
            triggerHeight,
            popoverHeight,
            clientHeight,
            ['top', 'bottom'],
        );
        popoverStyle.left = pos;

        // console.log('popoverStyle', popoverStyle.top, popoverStyle.left)

        return popoverStyle;
      }
    }

                                        $(function() {
                                            // 全局变量存储，用于垃圾回收
                                            var heightMonitor = {
                                                timeoutId: null,
                                                lastExecTime: 0,
                                                observer: null,
                                                isDestroyed: false,
                                                currentHeight: null
                                            };

                                            // 节流函数实现 (ES5)
                                            function throttle(func, delay) {
                                                return function() {
                                                    if (heightMonitor.isDestroyed) return;

                                                    var context = this;
                                                    var args = arguments;
                                                    var currentTime = new Date().getTime();

                                                    if (currentTime - heightMonitor.lastExecTime > delay) {
                                                        func.apply(context, args);
                                                        heightMonitor.lastExecTime = currentTime;
                                                    } else {
                                                        if (heightMonitor.timeoutId) {
                                                            clearTimeout(heightMonitor.timeoutId);
                                                        }
                                                        heightMonitor.timeoutId = setTimeout(function() {
                                                            if (!heightMonitor.isDestroyed) {
                                                                func.apply(context, args);
                                                                heightMonitor.lastExecTime = new Date().getTime();
                                                            }
                                                        }, delay - (currentTime - heightMonitor.lastExecTime));
                                                    }
                                                };
                                            }

                                            // 获取内容区域高度的函数 (ES5)
                                            function getContentHeight() {
                                                return Math.max(
                                                    document.body.scrollHeight || 0,
                                                    document.body.offsetHeight || 0,
                                                    document.documentElement.clientHeight || 0,
                                                    document.documentElement.scrollHeight || 0,
                                                    document.documentElement.offsetHeight || 0
                                                );
                                            }

                                            // 发送高度信息给指定域名的函数 (ES5)
                                            function sendHeightMessage() {
                                                if (heightMonitor.isDestroyed) return;

                                                var height = getContentHeight();

                                                // 进行高度差异比对，不一致时才发信息
                                                if (height === heightMonitor.currentHeight) return;

                                                heightMonitor.currentHeight = height;

                                                var message = {
                                                    type: 'contentHeightChange',
                                                    height: height,
                                                    timestamp: new Date().getTime(),
                                                    source: 'ibd-system'
                                                };

                                                // 发送给 check-jdsz.jd.com
                                                if (window.location.hostname === 'check-jdsz.jd.com') {
                                                    try {
                                                        window.parent && window.parent.postMessage(message, 'https://check-jdsz.jd.com');
                                                        // console.log('>> postMessage', message);
                                                    } catch(e) {
                                                        // if (window.console && console.warn) {
                                                        //     console.warn('Failed to send message to check-jdsz.jd.com:', e);
                                                        // }
                                                    }
                                                }


                                                // 发送给 jdsz.jd.com
                                                if (window.location.hostname === 'jdsz.jd.com') {
                                                    try {
                                                        window.parent && window.parent.postMessage(message, 'https://jdsz.jd.com');
                                                        // console.log('>> postMessage', message);
                                                    } catch(e) {
                                                        // if (window.console && console.warn) {
                                                        //     console.warn('Failed to send message to jdsz.jd.com:', e);
                                                        // }
                                                    }
                                                }
                                            }

                                            // 使用节流的高度变化处理函数
                                            var throttledSendHeight = throttle(sendHeightMessage, 300); // 300ms 节流

                                            // 监听窗口大小变化
                                            $(window).on('resize.heightMonitor', throttledSendHeight);

                                            // 监听DOM内容变化（使用MutationObserver）
                                            if (window.MutationObserver) {
                                                heightMonitor.observer = new MutationObserver(throttledSendHeight);
                                                heightMonitor.observer.observe(document.body, {
                                                    childList: true,
                                                    subtree: true,
                                                    attributes: true,
                                                    attributeFilter: ['style', 'class']
                                                });
                                            }

                                            // 监听图片加载完成
                                            $('img').on('load.heightMonitor', throttledSendHeight);

                                            // 监听内容滚动
                                            $(window).on('scroll.heightMonitor', throttledSendHeight);

                                            // 初始发送一次高度信息（添加定时器引用管理）
                                            var initTimeoutId = setTimeout(function() {
                                                if (!heightMonitor.isDestroyed) {
                                                    sendHeightMessage();
                                                }
                                                // 执行完毕后清理引用
                                                initTimeoutId = null;
                                            }, 100);


                                            // 垃圾回收函数
                                            function destroyHeightMonitor() {
                                                if (heightMonitor.isDestroyed) return;

                                                heightMonitor.isDestroyed = true;

                                                // 清理节流函数中的定时器
                                                if (heightMonitor.timeoutId) {
                                                    clearTimeout(heightMonitor.timeoutId);
                                                    heightMonitor.timeoutId = null;
                                                }

                                                // 清理初始化定时器
                                                if (initTimeoutId) {
                                                    clearTimeout(initTimeoutId);
                                                    initTimeoutId = null;
                                                }

                                                // 断开MutationObserver
                                                if (heightMonitor.observer) {
                                                    heightMonitor.observer.disconnect();
                                                    heightMonitor.observer = null;
                                                }

                                                // 移除事件监听器
                                                $(window).off('resize.heightMonitor scroll.heightMonitor');
                                                $('img').off('load.heightMonitor');

                                                // 清理对象引用
                                                heightMonitor = null;
                                            }

                                            // 页面卸载时进行垃圾回收
                                            $(window).on('beforeunload.heightMonitor', destroyHeightMonitor);

                                            // 暴露销毁方法到全局（可选，用于手动清理）
                                            if (window.ibdHeightMonitor) {
                                                window.ibdHeightMonitor.destroy = destroyHeightMonitor;
                                            } else {
                                                window.ibdHeightMonitor = {
                                                    destroy: destroyHeightMonitor
                                                };
                                            }
                                        });
